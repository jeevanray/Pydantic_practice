```python
"""Abstract base classes for database clients to ensure loose coupling."""
from abc import ABC, abstractmethod
from typing import Dict, Any, List, Tuple, Iterator, Optional
import pyarrow as pa
from contextlib import contextmanager

class DatabaseClient(ABC):
    """Abstract base class for database clients."""
    
    def __init__(self, config: Dict[str, Any]):
        self.config = config
    
    @abstractmethod
    def create_connection(self):
        """Create database connection."""
        pass
    
    @abstractmethod
    @contextmanager
    def connection(self):
        """Context manager for database connections."""
        pass
    
    @abstractmethod
    def execute_query(self, query: str, params: Dict[str, Any] = None) -> List[tuple]:
        """Execute a query and return results."""
        pass
    
    @abstractmethod
    def get_streaming_cursor(self, query: str, params: Dict[str, Any] = None, batch_size: int = 10000):
        """Get a streaming cursor for large result sets."""
        pass

class SchemaManager(ABC):
    """Abstract base class for schema management."""
    
    def __init__(self, db_client: DatabaseClient):
        self.db_client = db_client
    
    @abstractmethod
    def get_table_schema(self, table_name: str) -> Tuple[pa.Schema, Dict[str, pa.DataType]]:
        """Get table schema as PyArrow schema."""
        pass
    
    @abstractmethod
    def get_collection_schema(self, collection_name: str) -> Tuple[pa.Schema, Dict[str, pa.DataType]]:
        """Get collection schema as PyArrow schema (for MongoDB)."""
        pass
```

```python
"""Oracle database client implementation."""
import oracledb
from typing import Dict, Any, List, Tuple
from tenacity import retry, stop_after_attempt, wait_exponential, retry_if_exception_type
import pyarrow as pa
from contextlib import contextmanager

from .base_client import DatabaseClient, SchemaManager
from ..core.config import setup_logger
from ..core.exceptions import DatabaseConnectionError, SchemaError

logger = setup_logger(__name__)

class OracleClient(DatabaseClient):
    """Oracle database client implementation."""
    
    @retry(stop=stop_after_attempt(3), wait=wait_exponential(multiplier=1, min=2, max=10))
    def create_connection(self) -> oracledb.Connection:
        """Create Oracle database connection with retry logic."""
        logger.info("Connecting to Oracle DB")
        try:
            conn = oracledb.connect(
                user=self.config.get("username", self.config.get("user", "")),
                password=self.config["password"],
                dsn=self.config["dsn"],
            )
            logger.info("Oracle connection successful")
            return conn
        except Exception as e:
            logger.error("Failed to connect to Oracle database: %s", e)
            raise DatabaseConnectionError(f"Oracle connection failed: {e}")
    
    @contextmanager
    def connection(self):
        """Context manager for Oracle connections."""
        conn = None
        try:
            conn = self.create_connection()
            yield conn
        finally:
            if conn:
                try:
                    conn.close()
                except Exception as e:
                    logger.warning("Error closing Oracle connection: %s", e)
    
    def execute_query(self, query: str, params: Dict[str, Any] = None) -> List[tuple]:
        """Execute a query and return results."""
        with self.connection() as conn:
            with conn.cursor() as cur:
                cur.execute(query, params or {})
                return cur.fetchall()
    
    def get_streaming_cursor(self, query: str, params: Dict[str, Any] = None, batch_size: int = 10000):
        """Get a streaming cursor for large result sets."""
        conn = self.create_connection()
        cur = conn.cursor()
        cur.arraysize = batch_size
        cur.execute(query, params or {})
        return conn, cur

class OracleSchemaManager(SchemaManager):
    """Oracle schema manager implementation."""
    
    def get_table_schema(self, table_name: str) -> Tuple[pa.Schema, Dict[str, pa.DataType]]:
        """Get Oracle table schema as PyArrow schema."""
        schema_query = """
            SELECT COLUMN_NAME, DATA_TYPE, DATA_PRECISION, DATA_SCALE, NULLABLE
            FROM ALL_TAB_COLUMNS
            WHERE TABLE_NAME = UPPER(:table_name) AND OWNER = UPPER(:schema_name)
            ORDER BY COLUMN_ID
        """
        
        if '.' in table_name:
            schema_name, table_only = table_name.split('.', 1)
        else:
            schema_name = self.db_client.config.get('schema', 'PUBLIC')
            table_only = table_name

        fields = []
        with self.db_client.connection() as conn:
            with conn.cursor() as cur:
                cur.execute(schema_query, {
                    'table_name': table_only.upper(),
                    'schema_name': schema_name.upper()
                })
                
                for row in cur.fetchall():
                    col_name, data_type, precision, scale, nullable = row
                    arrow_type = self._map_oracle_to_arrow_type(data_type, precision, scale)
                    is_nullable = (nullable == 'Y')
                    fields.append(pa.field(col_name, arrow_type, nullable=is_nullable))

        schema = pa.schema(fields)
        type_map = {f.name: f.type for f in fields}
        logger.info(f"Extracted schema for {table_name}: {len(fields)} columns")
        return schema, type_map
    
    def get_collection_schema(self, collection_name: str) -> Tuple[pa.Schema, Dict[str, pa.DataType]]:
        """Not applicable for Oracle."""
        raise NotImplementedError("Oracle doesn't have collections")
    
    def _map_oracle_to_arrow_type(self, data_type: str, precision: int, scale: int) -> pa.DataType:
        """Map Oracle data types to PyArrow types."""
        if data_type in ('VARCHAR2', 'NVARCHAR2', 'CHAR', 'NCHAR'):
            return pa.string()
        elif data_type in ('CLOB', 'NCLOB', 'LONG'):
            return pa.string()
        elif data_type == 'NUMBER':
            if precision is None:
                return pa.float64()
            elif scale == 0 or scale is None:
                if precision <= 9:
                    return pa.int32()
                elif precision <= 18:
                    return pa.int64()
                else:
                    return pa.decimal128(precision, 0)
            else:
                return pa.decimal128(min(precision, 38), min(scale, 38))
        elif data_type == 'DATE':
            return pa.date32()
        elif data_type == 'TIMESTAMP':
            return pa.timestamp('us')
        elif data_type in ('RAW', 'LONG RAW', 'BLOB'):
            return pa.binary()
        elif data_type == 'BOOLEAN':
            return pa.bool_()
        else:
            logger.warning(f"Unknown Oracle data type '{data_type}', defaulting to string")
            return pa.string()
```

```python
"""MongoDB client implementation."""
from pymongo import MongoClient
from typing import Dict, Any, List, Tuple, Iterator
import pyarrow as pa
from contextlib import contextmanager

from .base_client import DatabaseClient, SchemaManager
from ..core.config import setup_logger
from ..core.exceptions import DatabaseConnectionError, SchemaError

logger = setup_logger(__name__)

class MongoDBClient(DatabaseClient):
    """MongoDB client implementation."""
    
    def create_connection(self) -> MongoClient:
        """Create MongoDB connection."""
        try:
            client = MongoClient(
                host=self.config.get("host", "localhost"),
                port=self.config.get("port", 27017),
                username=self.config.get("username"),
                password=self.config.get("password"),
                authSource=self.config.get("auth_source", "admin")
            )
            # Test connection
            client.admin.command('ping')
            logger.info("MongoDB connection successful")
            return client
        except Exception as e:
            logger.error("Failed to connect to MongoDB: %s", e)
            raise DatabaseConnectionError(f"MongoDB connection failed: {e}")
    
    @contextmanager
    def connection(self):
        """Context manager for MongoDB connections."""
        client = None
        try:
            client = self.create_connection()
            yield client
        finally:
            if client:
                client.close()
    
    def execute_query(self, collection_name: str, query: Dict[str, Any] = None, projection: Dict[str, Any] = None) -> List[Dict]:
        """Execute a MongoDB query and return results."""
        with self.connection() as client:
            db = client[self.config["database"]]
            collection = db[collection_name]
            return list(collection.find(query or {}, projection))
    
    def get_streaming_cursor(self, collection_name: str, query: Dict[str, Any] = None, batch_size: int = 10000):
        """Get a streaming cursor for large result sets."""
        client = self.create_connection()
        db = client[self.config["database"]]
        collection = db[collection_name]
        cursor = collection.find(query or {}).batch_size(batch_size)
        return client, cursor

class MongoDBSchemaManager(SchemaManager):
    """MongoDB schema manager implementation."""
    
    def get_collection_schema(self, collection_name: str) -> Tuple[pa.Schema, Dict[str, pa.DataType]]:
        """Infer MongoDB collection schema by sampling documents."""
        with self.db_client.connection() as client:
            db = client[self.db_client.config["database"]]
            collection = db[collection_name]
            
            # Sample documents to infer schema
            sample_docs = list(collection.aggregate([{"$sample": {"size": 100}}]))
            
            if not sample_docs:
                raise SchemaError(f"No documents found in collection {collection_name}")
            
            # Infer schema from samples
            field_types = {}
            for doc in sample_docs:
                for key, value in doc.items():
                    if key not in field_types:
                        field_types[key] = self._infer_arrow_type(value)
            
            fields = [pa.field(name, dtype, nullable=True) for name, dtype in field_types.items()]
            schema = pa.schema(fields)
            
            logger.info(f"Inferred schema for {collection_name}: {len(fields)} fields")
            return schema, field_types
    
    def get_table_schema(self, table_name: str) -> Tuple[pa.Schema, Dict[str, pa.DataType]]:
        """MongoDB uses collections, not tables."""
        return self.get_collection_schema(table_name)
    
    def _infer_arrow_type(self, value) -> pa.DataType:
        """Infer PyArrow type from MongoDB value."""
        if isinstance(value, str):
            return pa.string()
        elif isinstance(value, int):
            return pa.int64()
        elif isinstance(value, float):
            return pa.float64()
        elif isinstance(value, bool):
            return pa.bool_()
        elif isinstance(value, list):
            return pa.list_(pa.string())  # Default to list of strings
        elif isinstance(value, dict):
            return pa.string()  # Store as JSON string
        else:
            return pa.string()  # Default fallback
```

```python
"""Updated data extractor with loose coupling for multiple database types."""
from typing import Dict, Any, Optional
from datetime import datetime

from ..core.config import setup_logger, DATETIMEFORMAT, IST
from ..database.base_client import DatabaseClient, SchemaManager
from ..storage.minio_client import MinioClient
from ..storage.arrow_processor import ArrowProcessor
from ..audit.audit_manager import AuditManager
from ..utils.helpers import generate_object_path
from ..core.exceptions import ValidationError, DataExtractionError

logger = setup_logger(__name__)

class DataExtractor:
    """Database-agnostic data extraction processor."""
    
    def __init__(self, db_client: DatabaseClient, schema_manager: SchemaManager,
                 minio_config: Dict[str, Any], audit_config: Dict[str, Any], db_type: str):
        self.db_client = db_client
        self.schema_manager = schema_manager
        self.minio_client = MinioClient(minio_config)
        self.audit_manager = AuditManager(audit_config)
        self.arrow_processor = ArrowProcessor()
        self.db_type = db_type.lower()
    
    def extract_table_to_parquet(
        self,
        table_name: str,
        base_object_path: str,
        business_loaddt: str,
        *,
        chunk_size: int = 100_000,
        compression: str = "snappy",
        order_by: Optional[str] = None,
        restart_point: int = 0,
        load_type: str = 'delta',
        delta_column: Optional[str] = None,
        delta_column_value: Optional[str] = None,
        sub_folder: Optional[str] = None,
    ) -> None:
        """Extract data from database to MinIO parquet files."""
        
        # Validation
        self._validate_extraction_params(table_name, base_object_path, business_loaddt, load_type, delta_column_value)
        
        extraction_start = datetime.now(IST)
        effective_object_path = generate_object_path(
            base_object_path, business_loaddt, load_type, delta_column_value, sub_folder
        )
        
        # Initialize audit
        audit_log = self.audit_manager.prepare_audit_log(
            table_name, business_loaddt, delta_column_value, load_type, 
            extraction_start, effective_object_path
        )
        
        # Check if already completed
        if self.audit_manager.is_job_completed(audit_log):
            logger.info(f"Job already completed for {table_name}. Skipping.")
            return
        
        # Get schema (works for both Oracle tables and MongoDB collections)
        if self.db_type == "oracle":
            schema, _ = self.schema_manager.get_table_schema(table_name)
        elif self.db_type == "mongodb":
            schema, _ = self.schema_manager.get_collection_schema(table_name)
        else:
            raise DataExtractionError(f"Unsupported database type: {self.db_type}")
        
        # Build query/filter based on database type
        query_filter = self._build_query_filter(load_type, delta_column, delta_column_value, order_by, table_name)
        
        # Process data in chunks
        self._process_data_chunks(
            query_filter, schema, effective_object_path, table_name,
            chunk_size, compression, restart_point, audit_log, 
            load_type, delta_column_value, extraction_start
        )
    
    def _build_query_filter(self, load_type: str, delta_column: Optional[str], 
                           delta_column_value: Optional[str], order_by: Optional[str], table_name: str):
        """Build query/filter based on database type."""
        if self.db_type == "oracle":
            where_clause = self._build_where_clause(load_type, delta_column, delta_column_value)
            order_clause = f" ORDER BY {order_by}" if order_by else ""
            return f"SELECT * FROM {table_name}{where_clause}{order_clause}"
        
        elif self.db_type == "mongodb":
            mongo_filter = {}
            if load_type == 'historic' and delta_column and delta_column_value:
                mongo_filter[delta_column] = {"$eq": delta_column_value}
            return mongo_filter
        
        else:
            raise DataExtractionError(f"Unsupported database type: {self.db_type}")
    
    def _process_data_chunks(self, query_filter, schema, effective_object_path: str,
                           table_name: str, chunk_size: int, compression: str, 
                           restart_point: int, audit_log: Dict[str, Any], 
                           load_type: str, delta_column_value: Optional[str], 
                           extraction_start: datetime):
        """Process data in chunks and upload to MinIO."""
        
        if self.db_type == "oracle":
            conn, cur = self.db_client.get_streaming_cursor(query_filter, batch_size=max(10_000, min(chunk_size, 100_000)))
        elif self.db_type == "mongodb":
            conn, cur = self.db_client.get_streaming_cursor(table_name, query_filter, chunk_size)
        else:
            raise DataExtractionError(f"Unsupported database type: {self.db_type}")
        
        try:
            # Skip to restart point
            for _ in range(restart_point):
                if self.db_type == "oracle":
                    skipped = cur.fetchmany(chunk_size)
                elif self.db_type == "mongodb":
                    skipped = list(cur.limit(chunk_size))
                    cur = cur.skip(chunk_size)
                
                if not skipped:
                    break
            
            chunk_index = restart_point
            column_names = [field.name for field in schema]
            total_records = int(audit_log.get("total_records", 0))
            
            while True:
                if self.db_type == "oracle":
                    rows = cur.fetchmany(chunk_size)
                    if not rows:
                        break
                    # Convert Oracle rows to arrow table
                    table_chunk = self.arrow_processor.create_arrow_table_from_rows(
                        rows, column_names, schema
                    )
                
                elif self.db_type == "mongodb":
                    docs = list(cur.limit(chunk_size))
                    if not docs:
                        break
                    # Convert MongoDB documents to arrow table
                    table_chunk = self.arrow_processor.create_arrow_table_from_documents(
                        docs, column_names, schema
                    )
                    cur = cur.skip(chunk_size)
                
                # Generate object name
                object_name = self._generate_chunk_object_name(
                    effective_object_path, table_name, load_type, 
                    delta_column_value, chunk_index
                )
                
                # Upload to MinIO
                self.minio_client.upload_arrow_table_as_parquet(
                    table_chunk, object_name, compression
                )
                
                # Update audit
                total_records += table_chunk.num_rows
                self._update_chunk_audit(
                    audit_log, chunk_index, total_records, 
                    extraction_start, effective_object_path
                )
                
                chunk_index += 1
            
            # Final audit update
            self._finalize_audit(audit_log, extraction_start)
            
        finally:
            if cur:
                if self.db_type == "oracle":
                    cur.close()
            if conn:
                if self.db_type == "oracle":
                    conn.close()
                elif self.db_type == "mongodb":
                    conn.close()
```
