from datetime import datetime, date
from typing import Optional, Dict, Any

DATETIME_FORMAT = "%Y-%m-%d %H:%M:%S"
DATE_FORMAT = "%Y-%m-%d"

def parsedatetime(value: Optional[str], fmt: str = DATETIME_FORMAT, isdate: bool = False) -> Optional[Any]:
    """
    Parse a datetime or date string into datetime or date object.
    Returns None if input is None.
    """
    if value is None:
        return None
    if isinstance(value, (datetime, date)):
        # Already parsed
        return value
    try:
        parsed = datetime.strptime(value, fmt)
        if isdate:
            return parsed.date()
        return parsed
    except Exception:
        return None

def ensuretimefields_auditdata(auditdata: Dict[str, Any]) -> Dict[str, Any]:
    """
    Normalize datetime and date fields in auditdata dictionary so that
    auditdata fields are datetime/date objects or None (not strings).
    """
    datetime_fields = ['taskstartts', 'taskendts']
    date_fields = ['businessloaddt']

    for field in datetime_fields:
        val = auditdata.get(field)
        auditdata[field] = parsedatetime(val, fmt=DATETIME_FORMAT, isdate=False)

    for field in date_fields:
        val = auditdata.get(field)
        auditdata[field] = parsedatetime(val, fmt=DATE_FORMAT, isdate=True)

    # Other fields can remain as-is; do not convert datetime fields to string here.
    return auditdata
